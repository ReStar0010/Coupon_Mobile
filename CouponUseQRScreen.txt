        <div style={{fontWeight:900,fontSize:26,fontFamily:"'JetBrains Mono',monospace",color:C.fg}}>${amount}</div>
        <div style={{flex:1}}>
          <div style={{fontWeight:700,fontSize:14,color:C.fg}}>{store}</div>
          <div style={{fontSize:11,color:'rgba(51,51,51,0.65)'}}>現金折抵券 · 到期 {params.expires||'11/08'}</div>

          borderRadius:4,fontFamily:"'JetBrains Mono',monospace",fontSize:10,fontWeight:700,color:C.yellow}}>⚡ 7天</div>
      </div>
      <div style={{flex:1,position:'relative',overflow:'hidden',margin:'0 16px',borderRadius:8,

        {/* Corner brackets */}
        {[['top','left'],['top','right'],['bottom','left'],['bottom','right']].map(([v,h],i)=>(
          <div key={i} style={{position:'absolute',[v]:18,[h]:18,width:30,height:30,pointerEvents:'none'}}>

          <div style={{position:'absolute',inset:0,background:'rgba(0,0,0,0.82)',display:'flex',
            flexDirection:'column',alignItems:'center',justifyContent:'center',zIndex:20}}>
