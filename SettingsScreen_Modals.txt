      {showEdit      && <EditProfileModal  name={profileName} email={profileEmail} phone={profilePhone}
                          onSave={(n,e,p,nameChg,emailChg,phoneChg)=>{
                            setProfileName(n);setProfileEmail(e);setProfilePhone(p);
                            if(emailChg){setEmailVerified(false);setVerifyField('email');}
                            else if(phoneChg){setPhoneVerified(false);setVerifyField('phone');}
                          }}
                          onClose={()=>setShowEdit(false)}/>}
      {showLogout    && <LogoutConfirmModal onLogout={()=>{setShowLogout(false);nav('home');}} onClose={()=>setShowLogout(false)}/>}
      {showDelete    && <DeleteAccountModal onClose={()=>setShowDelete(false)}/>}
      {showTerms     && <LegalTextModal title="服務條款" onClose={()=>setShowTerms(false)}/>}
      {showPrivacy   && <LegalTextModal title="隱私政策" onClose={()=>setShowPrivacy(false)}/>}
      {feedbackType  && <FeedbackModal type={feedbackType} onClose={()=>setFeedbackType(null)}/>}
      {showBlocked   && <BlockedMerchantsModal onClose={()=>setShowBlocked(false)}/>}
      {verifyField   && <VerifyModal field={verifyField}
                          currentVal={verifyField==='email'?profileEmail:profilePhone}
                          onClose={()=>{if(verifyField==='email')setEmailVerified(true);else setPhoneVerified(true);setVerifyField(null);}}/>}
    </div>
  );
};
